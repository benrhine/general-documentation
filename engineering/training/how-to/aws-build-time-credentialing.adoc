In order for CI builds to execute they must be able to get Aws Credentials at runtime. By default aws makes unique credentials available to every build run. In order to access them in your buildspec file use the following shell commands

[source, shell]
----
commands:
    # Environment variables created here are available to the running lambda
    # Get AWS Credentials
    - echo "-------------------------- Generate Credentials for $EXTERNAL_ROLE_ARN ---------------------------------"
    - CREDENTIALS=$(aws sts assume-role --role-arn $EXTERNAL_ROLE_ARN --role-session-name codebuild-deploy-use1 --duration-seconds 3600)
    - export AWS_ACCESS_KEY_ID="$(echo ${CREDENTIALS} | jq -r '.Credentials.AccessKeyId')"
    - export AWS_SECRET_ACCESS_KEY="$(echo ${CREDENTIALS} | jq -r '.Credentials.SecretAccessKey')"
    - export AWS_SESSION_TOKEN="$(echo ${CREDENTIALS} | jq -r '.Credentials.SessionToken')"
    - export AWS_EXPIRATION=$(echo ${CREDENTIALS} | jq -r '.Credentials.Expiration')
    - echo "-------------------------- Generate Credentials for $EXTERNAL_ROLE_ARN Complete ---------------------------------"
----
