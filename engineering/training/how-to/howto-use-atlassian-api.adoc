----
curl -X POST \
-H "Authorization: Bearer YOUR-TOKEN" \
-H "Content-Type: application/json" \
-d '{
"name": "v1.2.3",
"description": "Build #456 from CI pipeline",
"project": "DPDS"
"projectId": :16301,
"released": false,
"releaseDate": "2024-01-15"
}' \
https://YOUR-URL.com/jira/rest/api/3/version
----

----
curl -X GET \
-H "Authorization: Bearer YOUR-TOKEN" \
-H "Content-Type: application/json" \
https://YOUR-URL.com/rest/api/3/project/DPDS
----

----
curl -X GET \
-H "Authorization: Bearer YOUR-TOKEN" \
-H "Content-Type: application/json" \
https://YOUR-URL.com/jira/rest/api/2/mypermissions?projectKey=XXXX
----
